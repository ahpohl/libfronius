45template <typename T> void packInteger(uint16_t *dest, T value) {
46 static_assert(std::is_integral_v<T>, "T must be an integral type");
47
48 constexpr int nBytes = sizeof(T);
49 constexpr int nWords = nBytes / 2;
50
51 unsigned char buf[nBytes];
52 std::memcpy(buf, &value, nBytes);
53
54 for (int i = 0; i < nWords; i++) {
55 unsigned char hi = buf[2 * i + 1];
56 unsigned char lo = buf[2 * i];
57 dest[i] = (static_cast<uint16_t>(hi) << 8) | lo;
58 }
59
60 std::reverse(dest, dest + nWords);
61}

84inline uint32_t modbus_get_uint32(const uint16_t *regs, bool word_swap = false,
85 bool byte_swap = false) {
86 uint16_t hi = regs[0], lo = regs[1];
87 if (byte_swap) {
88 hi = swapBytes16(hi);
89 lo = swapBytes16(lo);
90 }
91 uint32_t val = ((uint32_t)hi << 16) | lo;
92 if (word_swap)
93 val = ((val & 0xFFFF) << 16) | (val >> 16);
94 return val;
95}

116inline uint64_t modbus_get_uint64(const uint16_t *regs, bool word_swap = false,
117 bool byte_swap = false) {
118 uint64_t val = 0;
119 for (int i = 0; i < 4; i++) {
120 uint16_t w = regs[i];
121 if (byte_swap)
122 w = swapBytes16(w);
123 val |= (uint64_t)w << ((3 - i) * 16);
124 }
125 if (word_swap) {
126 val = ((val & 0xFFFFULL) << 48) | ((val & 0xFFFF0000ULL) << 16) |
127 ((val >> 16) & 0xFFFF0000ULL) | ((val >> 48) & 0xFFFFULL);
128 }
129 return val;
130}

163inline FroniusTypes::RemoteEndpoint getSocketInfo(int socket) {
164 struct sockaddr_storage addr;
165 socklen_t addr_len = sizeof(addr);
166
167 if (getpeername(socket, (struct sockaddr *)&addr, &addr_len) != 0) {
168 return {"unknown", 0};
169 }
170
171 char ip_str[INET6_ADDRSTRLEN];
172 int port = 0;
173
174 if (addr.ss_family == AF_INET) {
175 // IPv4
176 struct sockaddr_in *addr_in = (struct sockaddr_in *)&addr;
177 inet_ntop(AF_INET, &addr_in->sin_addr, ip_str, sizeof(ip_str));
178 port = ntohs(addr_in->sin_port);
179 } else if (addr.ss_family == AF_INET6) {
180 // IPv6
181 struct sockaddr_in6 *addr_in6 = (struct sockaddr_in6 *)&addr;
182 inet_ntop(AF_INET6, &addr_in6->sin6_addr, ip_str, sizeof(ip_str));
183 port = ntohs(addr_in6->sin6_port);
184 } else {
185 return {"unknown", 0};
186 }
187
188 return {ip_str, port};
189}

219template <typename T>
220std::expected<void, ModbusError> packToModbus(modbus_mapping_t *dest,
221 Register reg, T value) {
222
223 if (!dest) {
224 return std::unexpected(
225 ModbusError::custom(EINVAL, "Null modbus_mapping_t pointer"));
226 }
227 if (!dest->tab_registers) {
228 return std::unexpected(
229 ModbusError::custom(EINVAL, "modbus_mapping_t has null tab_registers"));
230 }
231
232 switch (reg.TYPE) {
233 case Register::Type::INT16:
234 if constexpr (std::is_integral_v<T>)
235 dest->tab_registers[reg.ADDR] = static_cast<uint16_t>(value);
236 break;
237 case Register::Type::UINT16:
238 if constexpr (std::is_integral_v<T>)
239 dest->tab_registers[reg.ADDR] = value;
240 break;
241 case Register::Type::UINT32:
242 if constexpr (std::is_integral_v<T>)
243 detail::packInteger<uint32_t>(&dest->tab_registers[reg.ADDR], value);
244 break;
245 case Register::Type::UINT64:
246 if constexpr (std::is_integral_v<T>)
247 detail::packInteger<uint64_t>(&dest->tab_registers[reg.ADDR], value);
248 break;
249 case Register::Type::FLOAT:
250 if constexpr (std::is_floating_point_v<T>) {
251 // Not modbus_set_float_abcd(): that writes raw wire bytes, but
252 // modbus_reply() serialises each tab_registers entry as a host-order
253 // uint16, so on a little-endian host the bytes end up swapped inside
254 // every word. Route the bit pattern through packInteger instead, which
255 // already produces the host-order words the reply path expects.
256 const float f = static_cast<float>(value);
257 uint32_t bits;
258 std::memcpy(&bits, &f, sizeof(bits));
259 detail::packInteger<uint32_t>(&dest->tab_registers[reg.ADDR], bits);
260 }
261 break;
262 case Register::Type::STRING: {
263 if constexpr (std::is_same_v<T, std::string>) {
264 size_t maxLength = reg.NB * 2;
265 if (value.length() > maxLength) {
266 return std::unexpected(ModbusError::custom(
267 EINVAL,
268 "String length {} exceeds maximum {} characters for register {}",
269 value.length(), maxLength, reg.describe()));
270 }
271
272 // Pack string into modbus mapping
273 for (size_t i = 0; i < value.length() / 2; i++) {
274 unsigned char hi = value[2 * i];
275 unsigned char lo = value[2 * i + 1];
276 dest->tab_registers[reg.ADDR + i] =
277 (static_cast<uint16_t>(hi) << 8) | lo;
278 }
279 if (value.length() % 2) {
280 dest->tab_registers[reg.ADDR + value.length() / 2] =
281 (static_cast<uint16_t>(value[value.length() - 1]) << 8);
282 }
283
284 // Zero out remaining registers
285 for (size_t i = (value.length() + 1) / 2; i < static_cast<size_t>(reg.NB);
286 i++) {
287 dest->tab_registers[reg.ADDR + i] = 0;
288 }
289 }
290 break;
291 }
292 case Register::Type::UNKNOWN:
293 default:
294 return std::unexpected(ModbusError::custom(
295 EINVAL, std::format("Unsupported target register type: {}",
296 Register::typeToString(reg.TYPE))));
297 }
298
299 return {};
300}

328inline std::expected<void, ModbusError> packToModbus(modbus_mapping_t *dest,
329 Register reg, Register sf,
330 double realValue,
331 int decimals) {
332 if (!dest) {
333 return std::unexpected(
334 ModbusError::custom(EINVAL, "Null modbus_mapping_t pointer"));
335 }
336 if (!dest->tab_registers) {
337 return std::unexpected(
338 ModbusError::custom(EINVAL, "modbus_mapping_t has null tab_registers"));
339 }
340
341 // encode float into integer register
342 switch (reg.TYPE) {
343 case Register::Type::INT16: {
344 int16_t value =
345 static_cast<int16_t>(std::round(realValue * std::pow(10, decimals)));
346 dest->tab_registers[reg.ADDR] = static_cast<uint16_t>(value);
347 break;
348 }
349 case Register::Type::UINT16: {
350 uint16_t value =
351 static_cast<uint16_t>(std::round(realValue * std::pow(10, decimals)));
352 dest->tab_registers[reg.ADDR] = value;
353 break;
354 }
355
356 case Register::Type::UINT32: {
357 uint32_t value =
358 static_cast<uint32_t>(std::round(realValue * std::pow(10, decimals)));
359 detail::packInteger<uint32_t>(&dest->tab_registers[reg.ADDR], value);
360 break;
361 }
362 case Register::Type::UINT64: {
363 uint64_t value =
364 static_cast<uint64_t>(std::round(realValue * std::pow(10, decimals)));
365 detail::packInteger<uint64_t>(&dest->tab_registers[reg.ADDR], value);
366 break;
367 }
368
369 default:
370 return std::unexpected(ModbusError::custom(
371 EINVAL,
372 std::format("Unsupported target register type for encoding float: {}",
373 Register::typeToString(reg.TYPE))));
374 }
375
376 // encode scale factor (scale factors are INT16 in SunSpec)
377 int16_t sf_value = -static_cast<int16_t>(decimals);
378 dest->tab_registers[sf.ADDR] = static_cast<uint16_t>(sf_value);
379
380 return {};
381}
